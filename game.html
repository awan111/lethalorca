// Global State Variables let walletOk = false; let userPublicKey = null; // Initialize Solana Connection const solanaConnection = new solanaWeb3.Connection( solanaWeb3.clusterApiUrl('devnet'), 'confirmed' ); // ============================================================ // 1. SILENT AUTO-CONNECT & PHANTOM EVENT LISTENERS // ============================================================ window.addEventListener('load', async () => { const provider = window.phantom?.solana; if (provider?.isPhantom) { // A. Listen for account changes (e.g., user switches wallet in Phantom) provider.on('accountChanged', (publicKey) => { if (publicKey) { handleConnected(publicKey); } else { // Handle disconnect event from wallet side disconnectUI(); } }); // B. Listen for manual disconnects from Phantom popup provider.on('disconnect', () => { disconnectUI(); }); // C. SILENT AUTO-CONNECT (onlyIfTrusted: true) try { const resp = await provider.connect({ onlyIfTrusted: true }); handleConnected(resp.publicKey); console.log('⚡ Auto-connected to Phantom silently!'); } catch (err) { // User is either not connected or hasn't trusted this dapp yet. // Silent error is expected here, do nothing so UI stays disconnected. console.log('User not auto-connected:', err.message); } } }); // ============================================================ // 2. CONNECT / DISCONNECT UI HANDLERS // ============================================================ // Manual Connect Button Trigger async function connectPhantom() { try { const provider = window.phantom?.solana; if (!provider) { alert('❌ Phantom wallet not found! Please install it from phantom.app'); return; } // Standard connect popup const resp = await provider.connect(); handleConnected(resp.publicKey); } catch (e) { alert('Wallet Connection Error: ' + e.message); } } // UI state update when connected function handleConnected(pubKey) { userPublicKey = pubKey; walletOk = true; const addrStr = pubKey.toString(); document.getElementById('walletDisconnected').style.display = 'none'; document.getElementById('walletConnected').style.display = 'block'; document.getElementById('walletAddr').textContent = addrStr.slice(0, 6) + '...' + addrStr.slice(-6); const walletBtn = document.getElementById('walletBtn'); walletBtn.classList.add('connected'); walletBtn.textContent = '✓ Connected'; } // Manual Disconnect Button Trigger function disconnect() { if (window.phantom?.solana) { window.phantom.solana.disconnect(); } disconnectUI(); } // UI state update when disconnected function disconnectUI() { walletOk = false; userPublicKey = null; document.getElementById('walletDisconnected').style.display = 'block'; document.getElementById('walletConnected').style.display = 'none'; const walletBtn = document.getElementById('walletBtn'); walletBtn.classList.remove('connected'); walletBtn.textContent = 'Connect Wallet'; }
